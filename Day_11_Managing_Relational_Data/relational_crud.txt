create table users(id serial primary key,
name text,
email text,
created_at timestamp default current_timestamp);

create table orders(id serial primary key,
user_id int references users(id),
amount int,
status text,
created_at timestamp default current_timestamp)

-- Why should orders not be stored inside the users table?

# Orders should not be stored in the users table because
# one user can have multiple orders, which violates normalization.
# Keeping orders separate improves scalability and data integrity.
